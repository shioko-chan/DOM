cmake_minimum_required(VERSION 4.0)

project(SKYMERGE VERSION 1.0.0 LANGUAGES C CXX CUDA)

set(LOG_LEVEL INFO)

# set(WANTED_BUILD_TYPE DEBUG)
set(WANTED_BUILD_TYPE RELEASE)

set(ENABLE_VISUALIZE_OUTPUT FALSE)
set(ENABLE_PARALLEL TRUE)
set(ENABLE_ASSERTION FALSE)

if(${WANTED_BUILD_TYPE} STREQUAL "DEBUG")
set(CMAKE_BUILD_TYPE Debug)
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -g")
elseif(${WANTED_BUILD_TYPE} STREQUAL "RELEASE")
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3")
else()
message(FATAL_ERROR "Unknown build type: ${WANTED_BUILD_TYPE}. Use 'DEBUG' or 'RELEASE'.")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_subdirectory(src)
